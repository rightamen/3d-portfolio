cmake_minimum_required(VERSION 3.20)
project(mrright_native VERSION 0.3.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
option(MR_BUILD_SERVER "Build the PostgreSQL C++ catalogue service" ${UNIX})
option(MR_BUILD_DESKTOP "Build the Windows native desktop client" ${WIN32})
option(MR_BUILD_MODEL_TESTS "Build native GLB parser tests" OFF)
set(MR_REVISION "development" CACHE STRING "Exact source revision")
add_library(mr_json INTERFACE)
target_include_directories(mr_json INTERFACE "${CMAKE_CURRENT_SOURCE_DIR}/third_party" "${CMAKE_CURRENT_SOURCE_DIR}/common")
if(MR_BUILD_SERVER)
 find_package(PostgreSQL REQUIRED)
 find_package(Threads REQUIRED)
 add_executable(mrright-catalogue server/main.cpp server/Catalogue.cpp)
 target_include_directories(mrright-catalogue PRIVATE third_party/httplib)
 target_link_libraries(mrright-catalogue PRIVATE mr_json PostgreSQL::PostgreSQL Threads::Threads)
 target_compile_definitions(mrright-catalogue PRIVATE MR_REVISION="${MR_REVISION}")
 target_compile_options(mrright-catalogue PRIVATE -Wall -Wextra -Wpedantic)
 install(TARGETS mrright-catalogue RUNTIME DESTINATION bin)
endif()
if(MR_BUILD_DESKTOP OR MR_BUILD_MODEL_TESTS)
 include(FetchContent)
 set(DRACO_TESTS OFF CACHE BOOL "" FORCE)
 set(DRACO_JS_GLUE OFF CACHE BOOL "" FORCE)
 set(DRACO_BUILD_EXECUTABLES OFF CACHE BOOL "" FORCE)
 set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
 if(MR_DRACO_ARCHIVE)
  set(DRACO_SOURCE_URL "${MR_DRACO_ARCHIVE}")
 else()
  set(DRACO_SOURCE_URL "https://codeload.github.com/google/draco/tar.gz/refs/tags/1.5.7")
 endif()
 FetchContent_Declare(draco URL "${DRACO_SOURCE_URL}" URL_HASH SHA256=bf6b105b79223eab2b86795363dfe5e5356050006a96521477973aba8f036fe1)
 FetchContent_MakeAvailable(draco)
 add_library(mr_model common/Glb.cpp)
 target_link_libraries(mr_model PUBLIC mr_json draco::draco)
 target_include_directories(mr_model PUBLIC "${CMAKE_CURRENT_BINARY_DIR}" "${CMAKE_CURRENT_BINARY_DIR}/_deps/draco-src/src" "${CMAKE_CURRENT_BINARY_DIR}/_deps/draco-build")
endif()
if(MR_BUILD_DESKTOP)
 foreach(option WEBP_BUILD_ANIM_UTILS WEBP_BUILD_CWEBP WEBP_BUILD_DWEBP WEBP_BUILD_GIF2WEBP WEBP_BUILD_IMG2WEBP WEBP_BUILD_VWEBP WEBP_BUILD_WEBPINFO WEBP_BUILD_WEBPMUX WEBP_BUILD_EXTRAS WEBP_BUILD_LIBWEBPMUX)
  set(${option} OFF CACHE BOOL "" FORCE)
 endforeach()
 if(MR_WEBP_ARCHIVE)
  set(WEBP_SOURCE_URL "${MR_WEBP_ARCHIVE}")
 else()
  set(WEBP_SOURCE_URL "https://codeload.github.com/webmproject/libwebp/tar.gz/refs/tags/v1.6.0")
 endif()
 FetchContent_Declare(libwebp URL "${WEBP_SOURCE_URL}" URL_HASH SHA256=93a852c2b3efafee3723efd4636de855b46f9fe1efddd607e1f42f60fc8f2136)
 FetchContent_MakeAvailable(libwebp)
 if(NOT WIN32)
  message(FATAL_ERROR "Use the Windows compiler or cmake/mingw64.cmake")
 endif()
 add_executable(mrright-studio WIN32 desktop/Main.cpp desktop/Input.cpp desktop/App.cpp desktop/StudioCopy.cpp desktop/Exhibition.cpp desktop/Showcase.cpp desktop/ShowcaseCanvas.cpp desktop/WorkspaceApi.cpp desktop/Workspace.cpp desktop/WorkspaceCanvas.cpp desktop/Canvas.cpp desktop/NativeTests.cpp desktop/StudioTests.cpp desktop/DesignTests.cpp desktop/UiTests.cpp desktop/Http.cpp desktop/Windows.cpp desktop/Viewer.cpp desktop/Studio.rc)
 target_link_libraries(mrright-studio PRIVATE mr_model webpdecoder comctl32 d2d1 dwrite windowscodecs winhttp comdlg32 shell32 ole32 uuid user32 gdi32 dwmapi uxtheme opengl32 winmm)
 target_include_directories(mrright-studio PRIVATE desktop "${CMAKE_CURRENT_BINARY_DIR}/_deps/libwebp-src/src")
 target_compile_definitions(mrright-studio PRIVATE UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN _WIN32_WINNT=0x0A00 MR_REVISION="${MR_REVISION}")
 if(MINGW)
  target_compile_options(mrright-studio PRIVATE -Wall -Wextra -Wno-missing-field-initializers)
  target_link_options(mrright-studio PRIVATE -municode -static -static-libgcc -static-libstdc++)
 endif()
 install(TARGETS mrright-studio RUNTIME DESTINATION .)
endif()
if(MR_BUILD_MODEL_TESTS)
 enable_testing()
 add_executable(mrright-model-tests tests/model_tests.cpp)
 target_link_libraries(mrright-model-tests PRIVATE mr_model)
 add_test(NAME model-parser COMMAND mrright-model-tests)
endif()
